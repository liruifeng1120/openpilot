#=============AMD==============
cd /data/openpilot
mkdir gpu && cd gpu
mkdir amd && cd amd
# 下载安装包:
wget https://repo.radeon.com/amdgpu-install/6.4.4/ubuntu/noble/amdgpu-install_6.4.60404-1_all.deb
# 安装驱动:
sudo dpkg -i amdgpu-install_6.4.60404-1_all.deb
# usecase不要使用默认，避免导致显卡异常
sudo amdgpu-install --usecase=opencl --accept-eula
sudo apt install clinfo darktable
# 增加当前用户到render和video组
sudo usermod -a -G render $LOGNAME
sudo usermod -a -G video $LOGNAME
# 检查opencl是否正确
clinfo -l
